## 二、核心数据模型

### 2.1 CableLink — 链路数据结构

```java
package com.virtualredstonewire.data;

public class CableLink {
    private final String id;          // SHA-512 哈希标识
    private final BlockPos from;      // 输入端坐标
    private final BlockPos to;        // 输出端坐标
    private final Direction toFace;   // 输出端接收面
    
    public CableLink(BlockPos from, BlockPos to, Direction toFace) {
        this.from = from.immutable();
        this.to = to.immutable();
        this.toFace = toFace;
        this.id = generateId(from, to, toFace);
    }
    
    // SHA-512 哈希生成
    public static String generateId(BlockPos from, BlockPos to, Direction face) {
        String input = from.getX() + "," + from.getY() + "," + from.getZ() + "|"
                     + to.getX() + "," + to.getY() + "," + to.getZ() + "|"
                     + face.getName();
        return Hashing.sha512().hashString(input, StandardCharsets.UTF_8).toString();
    }
    
    // getters ...
}
```

### 2.2 CableNetwork — 网络管理器（核心数据引擎）

```java
package com.virtualredstonewire.data;

public class CableNetwork {
    // SHA-512 → CableLink 映射
    private final Map<String, CableLink> links = new HashMap<>();
    
    // 正向邻接表：from BlockPos → Set<(to BlockPos, Direction)>
    private final Map<BlockPos, Set<Map.Entry<BlockPos, Direction>>> forwardAdj = new HashMap<>();
    
    // 反向邻接表：(to BlockPos, Direction) → Set<from BlockPos>
    private final Map<Map.Entry<BlockPos, Direction>, Set<BlockPos>> reverseAdj = new HashMap<>();
    
    // 合并邻接查询表：同时用于双向快速判断
    // 包含 from→(to,face) 和 (to,face)→from 的双向快速查找
    
    /**
     * 添加链路
     * 返回值: true=新建, false=删除(已存在)
     */
    public boolean toggleLink(BlockPos from, BlockPos to, Direction toFace) {
        String id = CableLink.generateId(from, to, toFace);
        
        if (links.containsKey(id)) {
            // 删除已有链路
            removeLink(id);
            return false;  // 表示删除
        } else {
            // 创建新链路
            CableLink link = new CableLink(from, to, toFace);
            links.put(id, link);
            
            // 更新正向邻接表
            forwardAdj.computeIfAbsent(from, k -> new HashSet<>())
                      .add(Map.entry(to, toFace));
            
            // 更新反向邻接表
            reverseAdj.computeIfAbsent(Map.entry(to, toFace), k -> new HashSet<>())
                      .add(from);
            
            return true;  // 表示新建
        }
    }
    
    /**
     * 删除指定链路
     */
    public void removeLink(String id) {
        CableLink link = links.remove(id);
        if (link == null) return;
        
        // 从正向邻接表移除
        Set<Map.Entry<BlockPos, Direction>> outEdges = forwardAdj.get(link.getFrom());
        if (outEdges != null) {
            outEdges.remove(Map.entry(link.getTo(), link.getToFace()));
            if (outEdges.isEmpty()) forwardAdj.remove(link.getFrom());
        }
        
        // 从反向邻接表移除
        Set<BlockPos> inNodes = reverseAdj.get(Map.entry(link.getTo(), link.getToFace()));
        if (inNodes != null) {
            inNodes.remove(link.getFrom());
            if (inNodes.isEmpty()) reverseAdj.remove(Map.entry(link.getTo(), link.getToFace()));
        }
    }
    
    /**
     * 删除所有与指定坐标相连的链路（剪刀操作，删点）
     */
    public int removeAllLinksAt(BlockPos pos) {
        int count = 0;
        
        // 删除所有以 pos 为输入的链路
        Set<Map.Entry<BlockPos, Direction>> outEdges = forwardAdj.get(pos);
        if (outEdges != null) {
            // 复制一份，避免 ConcurrentModification
            List<Map.Entry<BlockPos, Direction>> edges = new ArrayList<>(outEdges);
            for (Map.Entry<BlockPos, Direction> edge : edges) {
                String id = CableLink.generateId(pos, edge.getKey(), edge.getValue());
                removeLink(id);
                count++;
            }
        }
        
        // 删除所有以 pos 为输出的链路
        // 需要遍历所有可能的接收面
        for (Direction face : Direction.values()) {
            Map.Entry<BlockPos, Direction> key = Map.entry(pos, face);
            Set<BlockPos> inNodes = reverseAdj.get(key);
            if (inNodes != null) {
                List<BlockPos> nodes = new ArrayList<>(inNodes);
                for (BlockPos from : nodes) {
                    String id = CableLink.generateId(from, pos, face);
                    removeLink(id);
                    count++;
                }
            }
        }
        
        return count;
    }
    
    /**
     * 获取指定输出端的所有输入坐标
     */
    public Set<BlockPos> getInputsForOutput(BlockPos to, Direction toFace) {
        return reverseAdj.getOrDefault(Map.entry(to, toFace), Collections.emptySet());
    }
    
    /**
     * 获取指定输入端的所有输出目标
     */
    public Set<Map.Entry<BlockPos, Direction>> getOutputsForInput(BlockPos from) {
        return forwardAdj.getOrDefault(from, Collections.emptySet());
    }
    
    /**
     * 查询指定坐标是否直接连接（任何方向）
     */
    public boolean hasLinkAt(BlockPos pos) {
        return forwardAdj.containsKey(pos) || 
               reverseAdj.containsKey(Map.entry(pos, Direction.NORTH)) ||
               reverseAdj.containsKey(Map.entry(pos, Direction.SOUTH)) ||
               reverseAdj.containsKey(Map.entry(pos, Direction.EAST)) ||
               reverseAdj.containsKey(Map.entry(pos, Direction.WEST)) ||
               reverseAdj.containsKey(Map.entry(pos, Direction.UP)) ||
               reverseAdj.containsKey(Map.entry(pos, Direction.DOWN));
    }
    
    /**
     * 查询指定坐标的出边列表（GUI 用）
     */
    public List<String> queryOutgoing(BlockPos pos) {
        List<String> result = new ArrayList<>();
        Set<Map.Entry<BlockPos, Direction>> edges = forwardAdj.get(pos);
        if (edges != null) {
            for (Map.Entry<BlockPos, Direction> edge : edges) {
                result.add("→ " + edge.getKey() + " [" + edge.getValue() + "]");
            }
        }
        return result;
    }
    
    /**
     * 查询指定坐标的入边列表（GUI 用）
     */
    public List<String> queryIncoming(BlockPos pos) {
        List<String> result = new ArrayList<>();
        for (Direction face : Direction.values()) {
            Set<BlockPos> froms = reverseAdj.get(Map.entry(pos, face));
            if (froms != null) {
                for (BlockPos from : froms) {
                    result.add("← " + from + " [" + face + "]");
                }
            }
        }
        return result;
    }
    
    // 序列化为 JSON
    public JsonObject toJson() { ... }
    
    // 从 JSON 反序列化
    public static CableNetwork fromJson(JsonObject json) { ... }
    
    // 获取所有链路（客户端渲染用）
    public Collection<CableLink> getAllLinks() {
        return links.values();
    }
}
```

## 三、物品交互实现

### 3.1 VirtualCableItem — 线缆交互状态机

```java
package com.virtualredstonewire.item;

public class VirtualCableItem extends Item {
    
    public VirtualCableItem(Properties properties) {
        super(properties);
    }
    
    @Override
    public InteractionResult useOn(UseOnContext context) {
        Level level = context.getLevel();
        Player player = context.getPlayer();
        BlockPos clickedPos = context.getClickedPos();
        Direction clickedFace = context.getClickedFace();
        
        if (level.isClientSide) return InteractionResult.SUCCESS;
        
        // 获取玩家选中的输入端坐标（保存在 PlayerData 或 Capability 中）
        BlockPos selectedInput = getSelectedInput(player);
        
        if (selectedInput == null) {
            // 状态1: 未选中 — 选中输入端
            setSelectedInput(player, clickedPos);
            player.sendSystemMessage(
                Component.translatable("message.virtual-redstone-wire.selected_input", 
                    formatPos(clickedPos)));
            return InteractionResult.SUCCESS;
        }
        
        if (selectedInput.equals(clickedPos)) {
            // 状态2: 已选中，点击同一方块 — 取消选中
            clearSelectedInput(player);
            player.sendSystemMessage(
                Component.translatable("message.virtual-redstone-wire.cancelled"));
            return InteractionResult.SUCCESS;
        }
        
        // 状态3: 已选中，点击不同方块 — 创建/删除链路
        // 距离检查
        double distance = Math.sqrt(selectedInput.distSqr(clickedPos));
        if (distance > ModConfig.maxLinkDistance.get()) {
            player.sendSystemMessage(
                Component.translatable("message.virtual-redstone-wire.too_far"));
            return InteractionResult.FAIL;
        }
        
        CableNetwork network = CableNetworkManager.get(level);
        boolean created = network.toggleLink(selectedInput, clickedPos, clickedFace);
        
        // 标记脏数据
        CableNetworkManager.markDirty(level);
        
        // 发送更新包到客户端
        CableUpdatePacket.send(level, selectedInput, clickedPos, clickedFace, created);
        
        // 更新虚拟红石源方块
        if (created) {
            placeVirtualSource(level, clickedPos, clickedFace);
            player.sendSystemMessage(
                Component.translatable("message.virtual-redstone-wire.link_created",
                    formatPos(selectedInput), formatPos(clickedPos)));
        } else {
            removeVirtualSourceIfEmpty(level, clickedPos, clickedFace);
            player.sendSystemMessage(
                Component.translatable("message.virtual-redstone-wire.link_removed",
                    formatPos(selectedInput), formatPos(clickedPos)));
        }
        
        // 清空选中状态
        clearSelectedInput(player);
        
        return InteractionResult.SUCCESS;
    }
}
```

## 四、渲染系统

### 4.1 链路高亮渲染（放大镜手持时）

**渲染元素**：
1. **输入端方块**：淡蓝色 (#0096FF) 六面边框
2. **输出端接收面**：淡黄色 (#FFFF64) 单面四边形边框（仅接收面那一面）
3. **连接线**：淡红色 (#DC5050) 直线段
   - 起点：输入端方块中心 (pos.getX()+0.5, pos.getY()+0.5, pos.getZ()+0.5)
   - 终点：输出端方块接收面的面中心

**终点计算示例**（输出端方块的接收面中心）：
```java
private static Vector3d getFaceCenter(BlockPos pos, Direction face) {
    double cx = pos.getX() + 0.5;
    double cy = pos.getY() + 0.5;
    double cz = pos.getZ() + 0.5;
    
    switch (face) {
        case DOWN:  return new Vector3d(cx, pos.getY(), cz);
        case UP:    return new Vector3d(cx, pos.getY() + 1, cz);
        case NORTH: return new Vector3d(cx, cy, pos.getZ());
        case SOUTH: return new Vector3d(cx, cy, pos.getZ() + 1);
        case WEST:  return new Vector3d(pos.getX(), cy, cz);
        case EAST:  return new Vector3d(pos.getX() + 1, cy, cz);
    }
    return new Vector3d(cx, cy, cz);
}
```

### 4.4 渲染实现关键代码段

使用 Forge 的 `RenderLevelStageEvent`（1.20.1 Forge 标准管线钩子）：

```java
@Mod.EventBusSubscriber(modid = MOD_ID, bus = Bus.FORGE)
public class CableRenderer {
    
    @SubscribeEvent
    public static void onRenderLevel(RenderLevelStageEvent event) {
        if (event.getStage() != RenderLevelStageEvent.Stage.AFTER_TRANSLUCENT_BLOCKS) return;
        
        Player player = Minecraft.getInstance().player;
        if (player == null) return;
        
        ItemStack mainHand = player.getMainHandItem();
        ItemStack offHand = player.getOffhandItem();
        boolean holdingCable = mainHand.getItem() instanceof VirtualCableItem 
                            || offHand.getItem() instanceof VirtualCableItem;
        boolean holdingCutter = mainHand.getItem() instanceof CableCutterItem 
                             || offHand.getItem() instanceof CableCutterItem;
        boolean holdingMagnifier = mainHand.getItem() instanceof CableMagnifierItem 
                                || offHand.getItem() instanceof CableMagnifierItem;
        
        if (!holdingCable && !holdingCutter && !holdingMagnifier) return;
        
        Level level = player.level;
        CableNetwork network = CableNetworkManager.get(level);
        
        PoseStack poseStack = event.getPoseStack();
        MultiBufferSource.BufferSource bufferSource = Minecraft.getInstance().renderBuffers().bufferSource();
        Vec3 cameraPos = event.getCamera().getPosition();
        
        poseStack.pushPose();
        poseStack.translate(-cameraPos.x, -cameraPos.y, -cameraPos.z);
        
        double renderDistSq = ModConfig.magnifierRenderDistance.get() 
                            * ModConfig.magnifierRenderDistance.get();
        
        for (CableLink link : network.getAllLinks()) {
            Vec3 fromCenter = link.getFromCenter();
            // 距离裁剪
            if (fromCenter.distanceToSqr(cameraPos) > renderDistSq) continue;
            
            if (holdingMagnifier) {
                // 渲染输入端淡蓝色边框
                renderBlockOutline(poseStack, bufferSource, link.getFrom(), 
                    new float[]{0.0f, 0.59f, 1.0f, 0.6f});  // #0096FF
                
                // 渲染输出端接收面淡黄色边框
                renderFaceOutline(poseStack, bufferSource, link.getTo(), link.getToFace(),
                    new float[]{1.0f, 1.0f, 0.39f, 0.6f});  // #FFFF64
                
                // 渲染连接线（起点：中心，终点：接收面中心）
                renderLine(poseStack, bufferSource, 
                    link.getFromCenter(), link.getFaceCenter(),
                    new float[]{0.86f, 0.31f, 0.31f, 0.5f});  // #DC5050
            } else if (holdingCable || holdingCutter) {
                // 淡暗蓝色渲染输入端（所有已有链路的输入端点）
                renderBlockOutline(poseStack, bufferSource, link.getFrom(),
                    new float[]{0.4f, 0.5f, 0.8f, 0.3f});  // 淡暗蓝
            }
        }
        
        poseStack.popPose();
        bufferSource.endBatch();
    }
```

```java
        return 0;

    @Override
    public int getDirectSignal(BlockState state, BlockGetter level, BlockPos pos, Direction direction) {
        return getSignal(state, level, pos, direction);  // 强红石信号
    }

    // 即输出端方块的相邻位置，朝向输出端方块
```

### 5.3 红石计算引擎

```java
package com.virtualredstonewire.redstone;

public class RedstoneCalculator {
    
    // 脏输出集合
    private static final Set<Map.Entry<BlockPos, Direction>> DIRTY_OUTPUTS = new HashSet<>();
    
    /**
     * 标记输出端为脏（需要重新计算）
     */
    public static void markDirty(BlockPos toPos, Direction toFace) {
        DIRTY_OUTPUTS.add(Map.entry(toPos, toFace));
    }
    
    /**
     * 处理所有脏输出（在 ServerLevel tick 中调用）
     */
    public static void tick(ServerLevel level) {
        if (DIRTY_OUTPUTS.isEmpty()) return;
        
        CableNetwork network = CableNetworkManager.get(level);
        
        Set<Map.Entry<BlockPos, Direction>> toProcess = new HashSet<>(DIRTY_OUTPUTS);
        DIRTY_OUTPUTS.clear();
        
        for (Map.Entry<BlockPos, Direction> output : toProcess) {
            BlockPos toPos = output.getKey();
            Direction toFace = output.getValue();
            
            // 获取所有输入
            Set<BlockPos> inputs = network.getInputsForOutput(toPos, toFace);
            if (inputs.isEmpty()) continue;
            
            // 计算最大信号强度（或逻辑）
            int maxPower = 0;
            for (BlockPos inputPos : inputs) {
                int power = getInputPower(level, inputPos);
                if (power > maxPower) maxPower = power;
            }
            
            // 更新虚拟源方块
            BlockPos sourcePos = toPos.relative(toFace.getOpposite());
            BlockState state = level.getBlockState(sourcePos);
            if (state.getBlock() instanceof VirtualRedstoneSourceBlock) {
                int currentPower = state.getValue(VirtualRedstoneSourceBlock.POWER);
                if (currentPower != maxPower) {
                    level.setBlock(sourcePos, 
                        state.setValue(VirtualRedstoneSourceBlock.POWER, maxPower),
                        Block.UPDATE_ALL);
                    // 通知邻居红石更新
                    level.updateNeighborsAt(sourcePos, state.getBlock());
                }
            }
        }
    }
    
    /**
     * 获取输入端坐标的红石信号强度
     * 取该坐标所有方向的最大输入信号值
     */
    private static int getInputPower(Level level, BlockPos pos) {
        int maxPower = 0;
        BlockState state = level.getBlockState(pos);
        
        if (state.isRedstoneConductor(level, pos)) {
            // 如果是红石导体，获取其信号
            for (Direction dir : Direction.values()) {
                int power = level.getSignal(pos, dir);
                if (power > maxPower) maxPower = power;
            }
        } else {
            // 非导体，获取来自周围的红石信号
            for (Direction dir : Direction.values()) {
                BlockPos neighborPos = pos.relative(dir);
                BlockState neighborState = level.getBlockState(neighborPos);
                int power = neighborState.getSignal(level, neighborPos, dir.getOpposite());
                if (power > maxPower) maxPower = power;
            }
        }
        
        return maxPower;
    }
}
```

```java
package com.virtualredstonewire.data;

public class CableNetworkSavedData extends SavedData {
    private static final String DATA_NAME = "virtual-redstone-wire_network";

        this.network = network;
    }
    
    @Override

        // 将 network 序列化为 JSON 字符串存入 CompoundTag
        tag.putString("network", network.toJson().toString());
        return tag;
    }
    
    public static CableNetworkSavedData load(CompoundTag tag) {
        String json = tag.getString("network");
        CableNetwork network = CableNetwork.fromJson(
            new Gson().fromJson(json, JsonElement.class).getAsJsonObject()
        );
        return new CableNetworkSavedData(network);
    }
    
    public static CableNetworkSavedData get(ServerLevel level) {
        return level.getDataStorage()
            .computeIfAbsent(CableNetworkSavedData::load, 
                () -> new CableNetworkSavedData(new CableNetwork()), 
                DATA_NAME);
    }
}
```
